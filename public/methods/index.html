Odin by Example: Methods

Odin has no class-based methods. Instead, define a standalone procedure that takes the value as its first parameter. The syntax reads naturally: the receiver is just an ordinary parameter.

$ odin run methods.odin -file
area of rectangle: 15
scaled rectangle: 30 x 50
rectangle pointer: width 6 height 10
distance: 5
package main

import "core:fmt"
import "core:math"

// The receiver is a regular first parameter.
Rectangle :: struct {
	width:  f64,
	height: f64,
}

// Read-only: take the value by value.
area :: proc(r: Rectangle) -> f64 {
	return r.width * r.height
}

// Build a new rectangle from the receiver's values.
scale :: proc(r: Rectangle, factor: f64) -> Rectangle {
	return Rectangle{width = r.width * factor, height = r.height * factor}
}

// Use a pointer receiver to mutate in place.
double :: proc(r: ^Rectangle) {
	r.width *= 2
	r.height *= 2
}

// A "method" is just a normal procedure you call yourself.
dist :: proc(p, q: Point) -> f64 {
	return math.sqrt((p.x - q.x) * (p.x - q.x) + (p.y - q.y) * (p.y - q.y))
}

Point :: struct {
	x, y: f64,
}

main :: proc() {
	rect := Rectangle{width = 3, height = 5}

	// Call it like any other procedure.
	fmt.println("area of rectangle:", area(rect))

	// Scale returns a new rectangle; rect is unchanged.
	big := scale(rect, 10)
	fmt.println("scaled rectangle:", big.width, "x", big.height)

	// Mutating receiver needs a pointer.
	double(&rect)
	fmt.println("rectangle pointer: width", rect.width, "height", rect.height)

	// Methods compose across types too.
	a := Point{0, 0}
	b := Point{3, 4}
	fmt.println("distance:", dist(a, b))
}
